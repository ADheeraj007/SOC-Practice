    function decodeBase64Safe(str) {
      try {
        const clean = str.trim();
        if (!clean || clean.length % 4 === 1 || !/^[A-Za-z0-9+/=]+$/.test(clean)) {
          return '[Error: Invalid Base64 input]';
        }
        const bin = atob(clean);
        const bytes = new Uint8Array(bin.length);
        for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
        return new TextDecoder('utf-8', { fatal: false }).decode(bytes);
      } catch {
        return '[Error: Invalid Base64 input]';
      }
    }

    function encodeBase64Safe(str) {
      try {
        const bytes = new TextEncoder().encode(str);
        let bin = '';
        for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
        return btoa(bin);
      } catch {
        return '[Error: Unable to encode input]';
      }
    }

    function decodeHexSafe(hexStr) {
      try {
        const clean = hexStr.replace(/[^0-9a-fA-F]/g, '');
        if (!clean || clean.length % 2 !== 0) {
          return '[Error: Hex string must have an even number of hex digits]';
        }
        const bytes = new Uint8Array(clean.length / 2);
        for (let i = 0; i < clean.length; i += 2) {
          bytes[i / 2] = parseInt(clean.slice(i, i + 2), 16);
        }
        return new TextDecoder('utf-8', { fatal: false }).decode(bytes);
      } catch {
        return '[Error: Invalid Hex input]';
      }
    }

    function rot13(str) {
      return str.replace(/[a-zA-Z]/g, (c) => {
        const base = c <= 'Z' ? 65 : 97;
        return String.fromCharCode(((c.charCodeAt(0) - base + 13) % 26) + base);
      });
    }

    function SkillRadar({ skills }) {
      const size = 240;
      const cx = size / 2;
      const cy = size / 2;
      const radius = 78;
      const count = SKILL_DOMAINS.length;

      const getPoint = (index, ratio) => {
        const angle = (Math.PI * 2 * index) / count - Math.PI / 2;
        return {
          x: cx + Math.cos(angle) * radius * ratio,
          y: cy + Math.sin(angle) * radius * ratio
        };
      };

      const rings = [0.25, 0.5, 0.75, 1];
      const dataPoints = SKILL_DOMAINS.map((d, i) => {
        const val = Math.min(100, Math.max(0, skills[d.key] || 0));
        return getPoint(i, val / 100);
      });
      const polygonStr = dataPoints.map((p) => `${p.x.toFixed(1)},${p.y.toFixed(1)}`).join(' ');

      return (
        <div className="flex flex-col items-center">
          <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} className="overflow-visible">
            {rings.map((r) => {
              const pts = SKILL_DOMAINS.map((_, i) => {
                const p = getPoint(i, r);
                return `${p.x.toFixed(1)},${p.y.toFixed(1)}`;

                <polygon
                  key={r}
                  points={pts}
                  fill="none"
                  stroke="rgba(148, 163, 184, 0.18)"

        try {
          const saved = localStorage.getItem(STORAGE_KEY);
          if (saved) {
            const parsed = JSON.parse(saved);
            return {
              ...DEFAULT_PROGRESS,
              ...parsed,
              completedTasks: Array.isArray(parsed.completedTasks) ? parsed.completedTasks : [],
              completedRooms: Array.isArray(parsed.completedRooms) ? parsed.completedRooms : [],
              containedHosts: Array.isArray(parsed.containedHosts) ? parsed.containedHosts : [],
              sigmaRules: Array.isArray(parsed.sigmaRules) ? parsed.sigmaRules : [],
              skills: { ...DEFAULT_PROGRESS.skills, ...(parsed.skills || {}) }
            };
          }
        } catch {
          // ignore storage errors
        }
        return DEFAULT_PROGRESS;
      });

      const [view, setView] = useState('dashboard'); // 'dashboard' | 'lab' | 'skills'
      const [activeRoomId, setActiveRoomId] = useState(ROOMS[0].id);
      const [tierFilter, setTierFilter] = useState('All');
      const [searchQuery, setSearchQuery] = useState('');

      const [taskInputs, setTaskInputs] = useState({});
      const [taskFeedback, setTaskFeedback] = useState({});
      const [revealedHints, setRevealedHints] = useState({});
      const [demoAssist, setDemoAssist] = useState(false);
      const [certificateRoom, setCertificateRoom] = useState(null);
      const [confirmReset, setConfirmReset] = useState(false);

      useEffect(() => {
        try {
          localStorage.setItem(STORAGE_KEY, JSON.stringify(progress));
        } catch {
          // ignore storage errors
        }
      }, [progress]);

      const activeRoom = useMemo(
        () => ROOMS.find((r) => r.id === activeRoomId) || ROOMS[0],
        [activeRoomId]
      );

      const nextRecommendedRoom = useMemo(() => {
        return ROOMS.find((r) => !progress.completedRooms.includes(r.id)) || ROOMS[ROOMS.length - 1];
      }, [progress.completedRooms]);

      const rankInfo = useMemo(() => getRankForXP(progress.xp), [progress.xp]);

      const filteredRooms = useMemo(() => {
        return ROOMS.filter((r) => {
          const matchesTier = tierFilter === 'All' || r.tier === tierFilter;
          const q = searchQuery.trim().toLowerCase();
          const matchesQuery =
            !q ||
            r.title.toLowerCase().includes(q) ||
            r.id.toLowerCase().includes(q) ||
            r.category.toLowerCase().includes(q) ||
            r.mitre.some((m) => m.toLowerCase().includes(q));
          return matchesTier && matchesQuery;
        });
      }, [tierFilter, searchQuery]);

      const openRoom = (roomId) => {
        const target = ROOMS.find((r) => r.id === roomId) || ROOMS[0];
        setActiveRoomId(target.id);
        setOpenTaskId(target.tasks[0]?.id || null);
        setView('lab');
      };

        if (progress.completedTasks.includes(task.id)) {
          setTaskFeedback((prev) => ({
            ...prev,
            [task.id]: { ok: true, msg: 'Already completed! XP already awarded.' }
          }));
          return;
        }

        const raw = (taskInputs[task.id] || '').trim();
        if (!raw) {
          setTaskFeedback((prev) => ({
            ...prev,
            [task.id]: { ok: false, msg: 'Enter an answer or flag before submitting.' }
          }));
          return;
        }

        const normalizedGuess = raw.toLowerCase().replace(/\\+/g, '/');
        const isCorrect = task.accepted.some(
          (acc) => acc.trim().toLowerCase().replace(/\\+/g, '/') === normalizedGuess
        );

        if (!isCorrect) {
          setTaskFeedback((prev) => ({
            ...prev,
            [task.id]: { ok: false, msg: 'Not quite. Check the logs/terminal or use the Hint.' }
          }));
          return;
        }

        const updatedCompletedTasks = [...progress.completedTasks, task.id];
        const roomAllDone = room.tasks.every((t) => updatedCompletedTasks.includes(t.id));
        const newlyCompletedRoom = roomAllDone && !progress.completedRooms.includes(room.id);
        const updatedCompletedRooms = newlyCompletedRoom

        const gainedXP = task.xp + bonusXP;

          ...prev,
          xp: prev.xp + gainedXP,
          completedTasks: updatedCompletedTasks,
          completedRooms: updatedCompletedRooms,
          skills: nextSkills
        }));

          ...prev,
          [task.id]: {
            ok: true,
            msg: newlyCompletedRoom
              ? `Correct! +${gainedXP} XP (includes +60 XP Room Completion Bonus)`
              : `Correct! +${task.xp} XP`
          }
        }));

                </div>
                {confirmReset ? (
                  <div className="flex items-center gap-1.5 bg-slate-900 border border-rose-500/40 rounded-lg px-2 py-1 text-xs">
                    <span className="text-rose-300">Reset?</span>
                    <button
                      type="button"
                      onClick={handleResetProgress}
                      className="px-1.5 py-0.5 rounded bg-rose-500/20 text-rose-200 hover:bg-rose-500/30 font-semibold"
                    >
                      Yes
                    </button>
                    <button
                      type="button"
                      onClick={() => setConfirmReset(false)}
                      className="px-1.5 py-0.5 rounded bg-slate-800 text-slate-300 hover:bg-slate-700"
                    >
                      No
                    </button>
                  </div>
                ) : (
                  <button
                    type="button"
                    onClick={() => setConfirmReset(true)}
                    title="Reset saved progress"
                    className="p-2 rounded-lg bg-slate-900 border border-slate-800 text-slate-400 hover:text-rose-300 hover:border-rose-500/40 transition"
                  >
                    <RotateCcw className="w-4 h-4" />
                  </button>
                )}
              </div>
            </div>
          </header>

          {/* MAIN CONTENT */}
          <main className="flex-1 max-w-7xl w-full mx-auto px-4 py-6">
            {view === 'dashboard' && (
              <div className="space-y-6">
                {/* HERO BANNER */}
                <section className="rounded-2xl bg-gradient-to-br from-slate-900 via-slate-900 to-cyan-950/30 border border-slate-800 p-5 sm:p-7 flex flex-col lg:flex-row lg:items-center lg:justify-between gap-6">
                  <div className="space-y-2.5 max-w-2xl">
                    <div className="inline-flex items-center gap-2 px-2.5 py-1 rounded-full bg-cyan-500/10 border border-cyan-500/30 text-cyan-300 text-xs font-medium">
                      <Zap className="w-3.5 h-3.5" />
                      Beginner to Expert Scenarios • Educational Simulation
                    </div>
                    <h1 className="text-2xl sm:text-3xl font-bold tracking-tight text-white">
                      Practice Real-World SOC Triage, Log Hunting &amp; DFIR
                    </h1>
                    <p className="text-sm text-slate-300 leading-relaxed">
                      Work through structured multi-stage rooms with simulated SIEM telemetry, an interactive Linux/DFIR terminal, CyberChef-Lite decoding, threat intel lookups, and Sigma/EDR containment actions.
                    </p>
                  </div>

                  <div className="flex flex-col sm:flex-row lg:flex-col items-stretch sm:items-center lg:items-end gap-3 shrink-0">
                    <button
                      type="button"
                      onClick={() => openRoom(nextRecommendedRoom.id)}
                      className="inline-flex items-center justify-center gap-2 px-4 py-2.5 rounded-xl bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-semibold text-xs sm:text-sm transition shadow-lg shadow-cyan-500/15"
                    >
                      <Play className="w-4 h-4 fill-current" />
                      <span>Resume: {nextRecommendedRoom.id} — {nextRecommendedRoom.title}</span>
                    </button>
                    <div className="flex items-center gap-4 text-xs text-slate-400">
                      <span>
                        Rooms Done:{' '}
                        <strong className="text-slate-100">
                          {progress.completedRooms.length}/{ROOMS.length}
                        </strong>
                      </span>
                      <span>
                        Tasks Solved:{' '}
                        <strong className="text-slate-100">
                          {progress.completedTasks.length}/{ROOMS.reduce((acc, r) => acc + r.tasks.length, 0)}
                        </strong>
                      </span>
                    </div>
                  </div>
                </section>

                {/* FILTERS */}
                <section className="flex flex-col md:flex-row md:items-center justify-between gap-3">
                  <div className="flex flex-wrap items-center gap-1.5">
                    {['All', ...TIERS.map((t) => t.id)].map((t) => (
                      <button
                        key={t}
                        type="button"
                        onClick={() => setTierFilter(t)}
                        className={`px-3 py-1.5 rounded-lg text-xs font-medium border transition ${
                          tierFilter === t
                            ? 'bg-cyan-500/15 text-cyan-300 border-cyan-500/40'
                            : 'bg-slate-900/70 text-slate-400 border-slate-800 hover:text-slate-200'
                        }`}
                      >
                        {t === 'All' ? 'All Tiers' : t}
                      </button>
                    ))}
                  </div>

                  <div className="relative w-full md:w-72">
                    <Search className="w-4 h-4 text-slate-400 absolute left-3 top-1/2 -translate-y-1/2" />
                    <input
                      type="text"
                      value={searchQuery}
                      onChange={(e) => setSearchQuery(e.target.value)}
                      placeholder="Search room, MITRE ID, or topic..."
                      className="w-full pl-9 pr-3 py-2 rounded-lg bg-slate-900 border border-slate-800 text-xs text-slate-100 placeholder:text-slate-500 focus:outline-none focus:border-cyan-500"
                    />
                  </div>
                </section>

                {/* ROOM GRID */}
                <section className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
                  {filteredRooms.map((room) => {
                    const tierMeta = TIERS.find((t) => t.id === room.tier);
                    const style = TIER_STYLES[tierMeta?.accent || 'cyan'];
                    const solvedCount = room.tasks.filter((t) =>
                      progress.completedTasks.includes(t.id)
                    ).length;
                    const isCompleted = progress.completedRooms.includes(room.id);

                    return (
                      <div
                        key={room.id}
                        className={`rounded-xl bg-slate-900/80 border border-slate-800 ${style.border} p-4 flex flex-col justify-between gap-4 transition`}
                      >
                        <div className="space-y-2.5">
                          <div className="flex items-center justify-between gap-2">
                            <span className="font-mono text-xs font-semibold text-cyan-400">
                              {room.id}
                            </span>
                            <div className="flex items-center gap-1.5">
                              <span className={`text-[11px] px-2 py-0.5 rounded border ${style.badge}`}>
                                {room.tier}
                              </span>
                              {isCompleted && (
                                <span className="inline-flex items-center gap-1 text-[11px] px-2 py-0.5 rounded bg-emerald-500/15 text-emerald-300 border border-emerald-500/30">
                                  <CheckCircle2 className="w-3 h-3" />
                                  Done
                                </span>
                              )}
                            </div>
                          </div>

                          <h2 className="text-base font-bold text-white leading-snug">
                            {room.title}
                          </h2>
                          <p className="text-xs text-slate-400 leading-relaxed">
                            {room.summary}
                          </p>

                          <div className="flex flex-wrap gap-1.5 pt-1">
                            {room.mitre.map((tech) => (
                              <span
                                key={tech}
                                className="font-mono text-[10px] px-2 py-0.5 rounded bg-slate-950 text-slate-300 border border-slate-800"
                              >
                                {tech}
                              </span>
                            ))}
                          </div>
                        </div>

                        <div className="space-y-3 pt-2 border-t border-slate-800/80">
                          <div className="flex items-center justify-between text-[11px] text-slate-400">
                            <span>
                              Tasks: <strong className="text-slate-200">{solvedCount}/{room.tasks.length}</strong>
                            </span>
                            <span>{room.estTime}</span>
                            <span className="font-mono text-cyan-300">+{room.xp} XP</span>
                          </div>

                          <button
                            type="button"
                            onClick={() => openRoom(room.id)}
                            className="w-full py-2 px-3 rounded-lg bg-slate-800 hover:bg-cyan-500/20 text-slate-100 hover:text-cyan-200 border border-slate-700 hover:border-cyan-500/40 text-xs font-semibold flex items-center justify-center gap-1.5 transition"
                          >
                            <span>{solvedCount > 0 ? 'Continue Lab' : 'Launch Lab'}</span>
                            <ArrowRight className="w-3.5 h-3.5" />
                          </button>
                        </div>
                      </div>
                    );
                  })}
                </section>
              </div>
            )}

            {view === 'lab' && (
              <div className="space-y-4">
                {/* LAB HEADER BAR */}
                <div className="rounded-xl bg-slate-900/90 border border-slate-800 p-4 flex flex-wrap items-center justify-between gap-3">
                  <div className="space-y-1">
                    <div className="flex flex-wrap items-center gap-2">
                      <button
                        type="button"
                        onClick={() => setView('dashboard')}
                        className="text-xs text-cyan-400 hover:underline"
                      >
                        ← All Rooms
                      </button>
                      <span className="text-slate-600">/</span>
                      <span className="font-mono text-xs text-slate-400">{activeRoom.id}</span>
                      <span className="text-xs px-2 py-0.5 rounded bg-slate-800 text-slate-300 border border-slate-700">
                        {activeRoom.difficulty} • {activeRoom.category}
                      </span>
                    </div>
                    <h1 className="text-lg sm:text-xl font-bold text-white">{activeRoom.title}</h1>
                    <p className="text-xs text-slate-300 max-w-3xl">{activeRoom.briefing}</p>
                  </div>

                  <div className="flex items-center gap-2.5">
                    <label className="inline-flex items-center gap-2 px-3 py-1.5 rounded-lg bg-slate-950 border border-slate-800 text-xs text-slate-300 cursor-pointer">
                      <input
                        type="checkbox"
                        checked={demoAssist}
                        onChange={(e) => setDemoAssist(e.target.checked)}
                        className="rounded border-slate-700 bg-slate-900 text-cyan-500"
                      />
                      <span>Show Example Answers</span>
                    </label>
                    <select
                      value={activeRoom.id}
                      onChange={(e) => openRoom(e.target.value)}
                      className="px-2.5 py-1.5 rounded-lg bg-slate-950 border border-slate-800 text-xs text-slate-200"
                    >
                      {ROOMS.map((r) => (
                        <option key={r.id} value={r.id}>
                          {r.id}: {r.title}
                        </option>
                      ))}
                    </select>
                  </div>
                </div>

                {/* SPLIT PANE: TASKS LEFT + WORKBENCH RIGHT */}
                <div className="grid grid-cols-1 lg:grid-cols-12 gap-4 items-start">
                  <div className="lg:col-span-5 space-y-2.5">
                    {activeRoom.tasks.map((task, idx) => {
                      const isOpen = openTaskId === task.id;
                      const isSolved = progress.completedTasks.includes(task.id);
                      const fb = taskFeedback[task.id];
                      const hintVisible = !!revealedHints[task.id];

                      return (
                        <div
                          key={task.id}
                          className={`rounded-xl border transition overflow-hidden ${
                            isSolved
                              ? 'bg-emerald-950/20 border-emerald-500/30'
                              : isOpen
                              ? 'bg-slate-900 border-cyan-500/40'
                              : 'bg-slate-900/70 border-slate-800'
                          }`}
                        >
                          <button
                            type="button"
                            onClick={() => setOpenTaskId(isOpen ? null : task.id)}
                            className="w-full px-4 py-3 flex items-center justify-between gap-2 text-left"
                          >
                            <div className="flex items-center gap-2.5 min-w-0">
                              {isSolved ? (
                                <CheckCircle2 className="w-4 h-4 text-emerald-400 shrink-0" />
                              ) : (
                                <span className="w-5 h-5 rounded-full bg-slate-800 text-slate-300 text-[11px] font-mono flex items-center justify-center shrink-0">
                                  {idx + 1}
                                </span>
                              )}
                              <span className="text-xs sm:text-sm font-semibold text-white truncate">
                                {task.title}
                              </span>
                            </div>
                            <div className="flex items-center gap-2 shrink-0">
                              <span className="font-mono text-[11px] text-cyan-300">+{task.xp} XP</span>
                              {isOpen ? (
                                <ChevronDown className="w-4 h-4 text-slate-400" />
                              ) : (
                                <ChevronRight className="w-4 h-4 text-slate-400" />
                              )}
                            </div>
                          </button>

                          {isOpen && (
                            <div className="px-4 pb-4 pt-1 space-y-3 border-t border-slate-800/70">
                              <p className="text-xs text-slate-200 leading-relaxed">{task.prompt}</p>

                              {demoAssist && (
                                <div className="p-2 rounded bg-amber-500/10 border border-amber-500/30 text-[11px] font-mono text-amber-200">
                                  Example answer: {task.example}
                                </div>
                              )}

                              {hintVisible && (
                                <div className="p-2 rounded bg-indigo-500/10 border border-indigo-500/30 text-xs text-indigo-200">
                                  <strong>Hint:</strong> {task.hint}
                                </div>
                              )}

                              <div className="flex flex-col sm:flex-row gap-2">
                                <input
                                  type="text"
                                  value={taskInputs[task.id] ?? (isSolved ? task.answer : '')}
                                  onChange={(e) =>
                                    setTaskInputs((prev) => ({ ...prev, [task.id]: e.target.value }))
                                  }
                                  onKeyDown={(e) => {
                                    if (e.key === 'Enter') handleSubmitAnswer(activeRoom, task);
                                  }}
                                  placeholder="Enter answer..."
                                  className="flex-1 px-3 py-1.5 rounded-lg bg-slate-950 border border-slate-700 font-mono text-xs text-white placeholder:text-slate-500 focus:outline-none focus:border-cyan-500"
                                />
                                <button
                                  type="button"
                                  onClick={() => handleSubmitAnswer(activeRoom, task)}
                                  className="px-3.5 py-1.5 rounded-lg bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-semibold text-xs transition shrink-0"
                                >
                                  Submit
                                </button>
                              </div>

                              <div className="flex items-center justify-between">
                                <button
                                  type="button"
                                  onClick={() =>
                                    setRevealedHints((prev) => ({ ...prev, [task.id]: !prev[task.id] }))
                                  }
                                  className="inline-flex items-center gap-1 text-[11px] text-slate-400 hover:text-cyan-300"
                                >
                                  <HelpCircle className="w-3.5 h-3.5" />
                                  <span>{hintVisible ? 'Hide Hint' : 'Need a Hint?'}</span>
                                </button>
                                {progress.completedRooms.includes(activeRoom.id) && (
                                  <button
                                    type="button"
                                    onClick={() => setCertificateRoom(activeRoom)}
                                    className="text-[11px] text-emerald-400 hover:underline font-medium"
                                  >
                                    View Debrief Summary
                                  </button>
                                )}
                              </div>

                              {fb && (
                                <div
                                  className={`text-xs font-medium px-2.5 py-1.5 rounded border ${
                                    fb.ok
                                      ? 'bg-emerald-500/10 text-emerald-300 border-emerald-500/30'
                                      : 'bg-rose-500/10 text-rose-300 border-rose-500/30'
                                  }`}
                                >
                                  {fb.msg}
                                </div>
                              )}
                            </div>
                          )}
                        </div>
                      );
                    })}
                  </div>

                  <div className="lg:col-span-7">
                    <LabWorkbench
                      room={activeRoom}
                      progress={progress}
                      onToggleContain={handleToggleContain}
                      onAddSigmaRule={handleAddSigmaRule}
                    />
                  </div>
                </div>
              </div>
            )}

            {view === 'skills' && (
              <div className="space-y-6">
                <div className="grid grid-cols-1 lg:grid-cols-12 gap-5">
                  <div className="lg:col-span-5 rounded-xl bg-slate-900/80 border border-slate-800 p-5 flex flex-col items-center justify-between gap-4">
                    <div className="text-center">
                      <h2 className="text-base font-bold text-white">5-Domain Defensive Skill Matrix</h2>
                      <p className="text-xs text-slate-400">
                        Rank: <span className={`font-semibold ${rankInfo.current.color}`}>{rankInfo.current.name}</span> • {progress.xp} XP
                      </p>
                    </div>
                    <SkillRadar skills={progress.skills} />
                    <div className="w-full space-y-2">
                      {SKILL_DOMAINS.map((d) => {
                        const val = progress.skills[d.key] || 0;
                        return (
                          <div key={d.key} className="space-y-1">
                            <div className="flex justify-between text-xs">
                              <span className="text-slate-300">{d.label}</span>
                              <span className="font-mono text-cyan-300">{val}%</span>
                            </div>
                            <div className="h-1.5 bg-slate-800 rounded-full overflow-hidden">
                              <div className="h-full bg-cyan-400" style={{ width: `${val}%` }} />
                            </div>
                          </div>
                        );
                      })}
                    </div>
                  </div>

                  <div className="lg:col-span-7 rounded-xl bg-slate-900/80 border border-slate-800 p-5 space-y-4">
                    <div>
                      <h2 className="text-base font-bold text-white">Operator Badges</h2>
                      <p className="text-xs text-slate-400">
                        Unlock badges by completing rooms and solving lab challenges.
                      </p>
                    </div>
                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                      {BADGES.map((b) => {
                        const unlocked = b.rule(progress);
                        return (
                          <div
                            key={b.id}
                            className={`p-3.5 rounded-xl border flex items-start gap-3 ${
                              unlocked
                                ? 'bg-slate-950/90 border-cyan-500/35 text-slate-100'
                                : 'bg-slate-950/40 border-slate-800/80 text-slate-500 opacity-75'
                            }`}
                          >
                            <div
                              className={`w-9 h-9 rounded-lg flex items-center justify-center shrink-0 ${
                                unlocked
                                  ? 'bg-cyan-500/15 text-cyan-300 border border-cyan-500/30'
                                  : 'bg-slate-900 text-slate-600 border border-slate-800'
                              }`}
                            >
                              {renderBadgeIcon(b.icon, 'w-4 h-4')}
                            </div>
                            <div>
                              <div className="text-xs font-bold flex items-center gap-1.5">
                                <span>{b.title}</span>
                                {unlocked && (
                                  <span className="text-[10px] px-1.5 py-0.5 rounded bg-emerald-500/15 text-emerald-300">
                                    Unlocked
                                  </span>
                                )}
                              </div>
                              <p className="text-[11px] text-slate-400 mt-0.5">{b.desc}</p>
                            </div>
                          </div>
                        );
                      })}
                    </div>
                  </div>
                </div>
              </div>
            )}
          </main>

          {/* ROOM COMPLETION MODAL */}
          {certificateRoom && (
            <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex items-center justify-center p-4">
              <div className="max-w-md w-full rounded-2xl bg-slate-900 border border-cyan-500/40 p-6 space-y-4 shadow-2xl">
                <div className="flex items-center justify-between">
                  <div className="inline-flex items-center gap-2 text-emerald-400 text-xs font-semibold uppercase tracking-wider">
                    <Trophy className="w-4 h-4" />
                    Room Cleared
                  </div>
                  <button
                    type="button"
                    onClick={() => setCertificateRoom(null)}
                    className="p-1 rounded hover:bg-slate-800 text-slate-400"
                  >
                    <X className="w-4 h-4" />
                  </button>
                </div>

                <div>
                  <h3 className="text-xl font-bold text-white">{certificateRoom.title}</h3>
                  <p className="text-xs text-slate-400 mt-1">
                    All {certificateRoom.tasks.length} tasks in <span className="font-mono text-cyan-300">{certificateRoom.id}</span> completed.
                  </p>
                </div>

                <div className="p-3.5 rounded-xl bg-slate-950 border border-slate-800 space-y-1.5 text-xs">
                  <div className="flex justify-between">
                    <span className="text-slate-400">Operator Rank</span>
                    <span className={`font-semibold ${rankInfo.current.color}`}>{rankInfo.current.name}</span>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-400">Total XP</span>
                    <span className="font-mono text-cyan-300">{progress.xp} XP</span>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-400">MITRE Techniques</span>
                    <span className="font-mono text-slate-200">{certificateRoom.mitre.join(', ')}</span>
                  </div>
                </div>

                <div className="flex justify-end gap-2">
                  <button
                    type="button"
                    onClick={() => setCertificateRoom(null)}
                    className="px-4 py-2 rounded-lg bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-semibold text-xs transition"
                  >
                    Continue Training
                  </button>
                </div>
              </div>
            </div>
          )}
        </div>
      );
    }

    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<CyberOpsSOCSimulator />);
